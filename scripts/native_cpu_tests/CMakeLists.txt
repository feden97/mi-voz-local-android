cmake_minimum_required(VERSION 3.22)
project(mivoz_cpu_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${ROOT}/external/qwen3-tts.cpp/ggml" ggml)
add_executable(mivoz_cpu_test cpu_test.cpp)
target_include_directories(mivoz_cpu_test PRIVATE "${ROOT}/patches")
target_link_libraries(mivoz_cpu_test PRIVATE ggml ggml-base ggml-cpu)
enable_testing()
add_test(NAME cpu_runtime COMMAND mivoz_cpu_test)
